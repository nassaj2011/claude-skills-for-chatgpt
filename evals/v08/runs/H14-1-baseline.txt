def normalize_llm_payload(payload):
    if not isinstance(payload, dict):
        return {"ok": False, "intent": None, "error": "INVALID"}

    if set(payload) != {"decision", "reason", "tool"}:
        return {"ok": False, "intent": None, "error": "INVALID"}

    decision = payload["decision"]
    reason = payload["reason"]
    tool = payload["tool"]

    if not isinstance(decision, str) or not isinstance(reason, str) or not isinstance(tool, str):
        return {"ok": False, "intent": None, "error": "INVALID"}

    if decision not in {"answer", "refuse", "lookup"} or tool not in {"none", "lookup_venue"}:
        return {"ok": False, "intent": None, "error": "INVALID"}

    if len(reason) > 120:
        return {"ok": False, "intent": None, "error": "INVALID"}

    if (decision == "lookup" and tool != "lookup_venue") or (
        decision != "lookup" and tool != "none"
    ):
        return {"ok": False, "intent": None, "error": "INVALID"}

    return {"ok": True, "intent": payload, "error": None}
