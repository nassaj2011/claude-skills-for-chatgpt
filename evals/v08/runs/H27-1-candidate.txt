def assess_package(metadata):
    reasons = []

    if not isinstance(metadata, dict):
        return {"decision": "REVIEW", "reasons": ["metadata must be a dict"]}

    required = {
        "name",
        "version",
        "source",
        "integrity",
        "license",
        "install_scripts",
    }

    missing = required - set(metadata)
    extra = set(metadata) - required

    if missing:
        reasons.append("missing required keys: " + ", ".join(sorted(missing)))
    if extra:
        reasons.append("unexpected keys: " + ", ".join(sorted(extra)))

    name = metadata.get("name")
    if not isinstance(name, str) or not name:
        reasons.append("name must be a nonempty string")

    version = metadata.get("version")
    valid_version = isinstance(version, str)
    if valid_version:
        parts = version.split(".")
        valid_version = (
            len(parts) == 3
            and all(part.isdigit() and part != "" for part in parts)
        )
    if not valid_version:
        reasons.append("version must be an exact numeric X.Y.Z version")

    if metadata.get("source") != "approved-registry":
        reasons.append("source must be approved-registry")

    integrity = metadata.get("integrity")
    hex_chars = "0123456789abcdef"
    if (
        not isinstance(integrity, str)
        or not integrity.startswith("sha256:")
        or len(integrity) != 71
        or any(char not in hex_chars for char in integrity[7:])
    ):
        reasons.append("integrity must be a sha256 hash with 64 lowercase hex characters")

    license_name = metadata.get("license")
    if not isinstance(license_name, str) or not license_name:
        reasons.append("license must be a nonempty string")

    if metadata.get("install_scripts") is not False:
        reasons.append("install_scripts must be False")

    return {
        "decision": "ALLOW" if not reasons else "REVIEW",
        "reasons": reasons,
    }
